\newcommand{\resumeItem}[1]{
  \item\small{
    {#1 \vspace{-2pt}}
  }
}

\newcommand{\resumeSubheading}[4]{
  \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & {\footnotesize #2} \\
      \textit{\small#3} & \textit{\footnotesize #4} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeSubSubheading}[2]{
    \item
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \textit{\small#1} & \textit{\footnotesize #2} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeProjectHeading}[2]{
    \item
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \small#1 & {\footnotesize #2} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeSubItem}[1]{\resumeItem{#1}\vspace{-4pt}}

\renewcommand\labelitemii{$\vcenter{\hbox{\tiny$\bullet$}}$}

\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=0.15in, label={}]}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}
\newcommand{\resumeItemListStart}{\begin{itemize}}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace{-10pt}}

%----------HEADING----------
\begin{center}
    \textbf{\Huge \scshape (( header.name | le ))} \\ \vspace{1pt}
    \small (( header.phone | le )) $|$ \href{mailto:(( header.email ))}{\underline{(( header.email | le ))}}
(% for link in header.links %)
    $|$ \href{(( link.url ))}{\underline{(( link.url | replace("https://", "") | replace("http://", "") | le ))}}
(% endfor %)
\end{center}

%-----------EDUCATION-----------
\section{Education}
  \resumeSubHeadingListStart
(% for edu in resume_content.education %)
    \resumeSubheading
      {(( edu.institution | le ))}{(( edu.startDate | le )) -- (( edu.endDate | le ))}
      {(( edu.degree | le ))}{(( edu.location | le if edu.location else "" ))}
(% endfor %)
  \resumeSubHeadingListEnd

%-----------TECHNICAL SKILLS-----------
\section{Technical Skills}
 \begin{itemize}[leftmargin=0.15in, label={}]
    \small{\item{
(% for skill in resume_content.skills %)
     \textbf{(( skill.category | le ))}{: (% for item in skill.items %)(( item | le ))(% if not loop.last %), (% endif %)(% endfor %)}(% if not loop.last %) \\(% endif %)
(% endfor %)
    }}
 \end{itemize}

%-----------EXPERIENCE-----------
\section{Experience}
  \resumeSubHeadingListStart
(% for exp in resume_content.work_experience %)
    \resumeSubheading
      {(( exp.title | le ))}{(( exp.startDate | le )) -- (( exp.endDate | le ))}
      {(( exp.company | le ))}{(( exp.location | le if exp.location else "" ))}
      \resumeItemListStart
(% for resp in exp.responsibilities %)
        \resumeItem{(( resp | le ))}
(% endfor %)
      \resumeItemListEnd
(% endfor %)
  \resumeSubHeadingListEnd

%-----------PROJECTS-----------
\section{Projects}
    \resumeSubHeadingListStart
(% for proj in resume_content.projects %)
        \resumeSubheading
          {\textbf{(( proj.name | le ))}}{(( proj.date | le if proj.date else "" ))}
          {(% if proj.tech_stack %)(( proj.tech_stack | le ))(% endif %)}{}
          \resumeItemListStart
(% for bullet in proj.description %)
            \resumeItem{(( bullet | le ))}
(% endfor %)
          \resumeItemListEnd
(% endfor %)
    \resumeSubHeadingListEnd
